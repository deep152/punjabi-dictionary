from nltk.corpus import wordnet as wn
from nltk.corpus import stopwords

def clean_definition(text: str) -> str:
    """
    Clean and standardize definition text.
    """
    text = re.sub(r'\([^)]*\)', '', text)
    text = text.replace(',', ';')
    text = ' '.join(text.split())
    return text.lower()

def get_synonyms(definition: str, pos: str = None) -> Set[str]:
    """
    Get WordNet synonyms filtered by part of speech.
    
    Args:
        definition (str): Clean definition text
        pos (str): Part of speech filter ('n' for noun, 'v' for verb, 
                  'a' for adjective, 'r' for adverb)
    """
    # Get English stopwords
    stop_words = set(stopwords.words('english'))
    
    # Split definition and remove stopwords
    words = definition.split()
    words = [word for word in words if word not in stop_words]
    
    synonyms = set()
    for word in words:
        # Get all synsets (synonym sets) for the word
        synsets = wn.synsets(word)
        
        # Filter by part of speech if specified
        if pos:
            synsets = [syn for syn in synsets if syn.pos() == pos]
            
        # Get all lemmas (word forms) from the synsets
        for synset in synsets:
            for lemma in synset.lemmas():
                synonyms.add(lemma.name())
    
    return synonyms

def find_possible_synonyms(df: pd.DataFrame, synonyms: Set[str], pos: str = None) -> List[str]:
    """
    Find possible Gurmukhi synonyms.
    
    Args:
        df: Dictionary DataFrame
        synonyms: Set of English synonyms
        pos: Part of speech to filter by
    """
    possible_synonyms = []
    
    for synonym in synonyms:
        # Find matches in definitions
        matching_rows = df[df['definition'].str.contains(synonym, case=False)]
        
        # Filter by part of speech if specified
        if pos:
            matching_rows = matching_rows[matching_rows['pos'].str.startswith(pos)]
        
        if not matching_rows.empty:
            gurmukhi_words = matching_rows['word'].values
            possible_synonyms.extend(gurmukhi_words)
    
    # Return unique synonyms, limited to top 5
    return list(dict.fromkeys(possible_synonyms))[:5]

def main():
    df = pd.read_csv('punjabi_university_dictionary.csv')
    
    gurmukhi_word = input("Enter a gurmukhi word: ")
    row = df[df['word'] == gurmukhi_word]
    
    if not row.empty:
        definition = row['definition'].values[0]
        pos = row['pos'].values[0]
        
        print(f"\nDefinition of {gurmukhi_word}: {definition}")
        
        # Clean definition
        clean_def = clean_definition(definition)
        
        # Map dictionary POS to WordNet POS codes
        pos_map = {
            'noun': 'n',
            'verb': 'v',
            'adjective': 'a',
            'adverb': 'r'
        }
        pos_code = next((v for k, v in pos_map.items() if k in pos.lower()), None)
        
        # Get and filter synonyms
        synonyms = get_synonyms(clean_def, pos_code)
        matches = find_possible_synonyms(df, synonyms, pos_code)
        
        print(f"\nSynonyms of {gurmukhi_word}:")
        for i, match in enumerate(matches, 1):
            match_def = df[df['word'] == match]['definition'].values[0]
            print(f"Syn {i}: {match}")
            print(f"Definition: {match_def}\n")
    else:
        print(f"The word '{gurmukhi_word}' was not found in the dictionary.")

if __name__ == "__main__":
    main()
